# LangGraph의 상태를 나타내는 딕셔너리의 키와 타입을 정의하는 TypedDict입니다.
class GraphState(TypedDict, total=False):
    question: Optional[str]  # 사용자의 초기 질문을 저장하는 변수입니다.
    db_context: Optional[str]  # Milvus에서 검색된 문서 내용을 저장하는 변수입니다.
    web_context: Optional[str]  # Tavily 웹 검색 결과를 저장하는 변수입니다.
    context: Optional[str]  # DB와 웹 검색 결과가 결합된 최종 컨텍스트를 저장하는 변수입니다.
    answer: Optional[str]  # 사용자에게 제공될 최종 답변을 저장하는 변수입니다.
    answer_draft: Optional[str]  # LLM이 생성한 초안 답변을 저장하는 변수입니다.
    answer_source: Optional[str]  # 답변의 출처(예: "내부 DB/웹 검색")를 저장하는 변수입니다.
    is_retrieval_sufficient: bool  # DB 검색 결과가 충분한지 여부(LLM 검증)를 저장하는 변수입니다.
    is_answer_sufficient: bool  # 생성된 답변이 충분한지 여부(LLM 검증)를 저장하는 변수입니다.


# ==================== 노드 함수들 ====================
# [노드] Milvus 벡터스토어에 연결하고 상태를 초기화하는 노드
def load_milvus_node(state: GraphState) -> Dict[str, Any]:
    """Milvus 벡터스토어에 연결하고 상태를 초기화하는 노드 함수"""
    print("\n--- 노드: Milvus 로드 ---")  # 현재 실행 중인 노드명을 출력합니다.
    global _vectorstore  # 전역 변수 _vectorstore를 사용하겠다고 선언합니다.
    # 'default' 별칭의 Milvus 연결이 존재하지 않으면 새로 연결을 시도합니다.
    if "default" not in connections.list_connections() or not connections.has_connection("default"):
        connections.connect(alias="default", uri=MILVUS_URI, token=MILVUS_TOKEN)
    try:
        # MilvusVectorStore 인스턴스를 생성하고 전역 변수에 할당합니다.
        _vectorstore = MilvusVectorStore(
            embedding_function=embedding_model,  # 사용할 임베딩 모델을 지정합니다.
            collection_name=MILVUS_COLLECTION,  # 연결할 컬렉션 이름을 지정합니다.
            connection_args={"uri": MILVUS_URI, "token": MILVUS_TOKEN},  # 연결 매개변수를 전달합니다.
        )
        print(f"✅ Milvus 로드 완료: {MILVUS_COLLECTION}")  # 성공 메시지를 출력합니다.
        return {**state}  # 현재 상태를 그대로 반환합니다.
    except Exception as e:
        print(f"❌ Milvus 로드 실패: {e}")  # 오류 메시지를 출력합니다.
        raise ConnectionError("Milvus 벡터스토어 로드 실패")  # Milvus 연결 실패 예외를 발생시킵니다.

# [노드] 사용자 질문에 기반하여 Milvus에서 관련 문서를 검색하는 노드
def retrieve_node(state: GraphState) -> Dict[str, Any]:
    """사용자 질문에 기반하여 Milvus에서 관련 문서를 검색하는 노드 함수"""
    print("--- 노드: 문서 검색 ---")  # 현재 노드명을 출력합니다.
    question = state.get("question")  # 상태에서 'question' 값을 가져옵니다.
    global _vectorstore  # 전역 변수 _vectorstore를 사용하겠다고 선언합니다.
    vectorstore = _vectorstore  # 전역 변수에서 벡터스토어 인스턴스를 가져옵니다.
    # 질문이나 벡터스토어 객체가 없으면 오류를 발생시킵니다.
    if not question or not vectorstore: raise ValueError("질문 또는 벡터스토어가 누락되었습니다.")
    
    # 더 나은 검색을 위해 질문에 키워드를 추가합니다.
    enhanced_query = f"{question} 재배 방법 키우기 팁"
    # Milvus에서 유사도 점수와 함께 상위 10개의 문서를 검색합니다.
    docs_with_scores = vectorstore.similarity_search_with_score(enhanced_query, k=10)
    
    # 검색된 문서 중 특정 기준(점수 0.5 이상, 내용 길이 100자 이상)을 만족하는 문서만 필터링합니다.
    filtered_docs = []
    for doc, score in docs_with_scores:
        content = doc.page_content or ""
        if score > 0.5 and len(content.strip()) > 100:
            filtered_docs.append(doc)
    
    # 최종적으로 사용할 상위 5개의 문서를 선택합니다.
    final_docs = filtered_docs[:5]

    context = ""
    if final_docs:
        print(f"✅ {len(final_docs)}개 문서 검색 완료.")  # 검색된 문서 수를 출력합니다.
        for i, doc in enumerate(final_docs):
            preview = (doc.page_content or "")[:100].replace("\n", " ")
            print(f"  - 문서 {i+1}: '{preview}...'")  # 각 문서의 일부 내용을 출력하여 확인합니다.
            context += f"\n\n{doc.page_content}"  # 모든 문서 내용을 하나의 문자열로 결합합니다.
    else:
        print("⚠️ 검색된 문서가 없습니다.")  # 검색된 문서가 없을 경우 경고를 출력합니다.
        context = "관련 문서를 찾을 수 없습니다."  # 컨텍스트를 해당 메시지로 설정합니다.
        
    print(f"  - 컨텍스트 길이: {len(context)}자")  # 최종 컨텍스트의 문자 길이를 출력합니다.
    return {**state, "db_context": context}  # 검색된 컨텍스트를 상태에 저장하고 반환합니다.

# [노드] LLM을 사용하여 DB 검색 결과가 충분한지 검증하는 노드
def retrieval_validation_node(state: GraphState) -> Dict[str, Any]:
    """LLM을 사용하여 DB 검색 결과가 질문에 답변하기에 충분한지 자체 검증하는 노드 함수"""
    print("--- 노드: 1차 검증 (검색 품질) ---")  # 현재 노드명을 출력합니다.
    question = state.get("question")  # 상태에서 'question' 값을 가져옵니다.
    db_context = state.get("db_context", "")  # 상태에서 'db_context' 값을 가져옵니다.
    
    # 검색된 문서가 없으면 불충분으로 판단하고 웹 검색으로 넘어갑니다.
    if "관련 문서를 찾을 수 없습니다." in db_context:
        print("⚠️ 검색된 문서가 없어 불충분으로 판단합니다.")  # 경고 메시지를 출력합니다.
        return {**state, "is_retrieval_sufficient": False}  # 상태에 검증 결과를 저장하고 반환합니다.

    # LLM을 사용하여 검증 프롬프트와 컨텍스트를 결합하는 체인(chain)을 만듭니다.
    validation_chain = ChatPromptTemplate.from_template(VALIDATION_PROMPT_TMPL) | llm | StrOutputParser()
    
    try:
        # LLM을 호출하여 'True' 또는 'False'를 응답으로 받습니다.
        response = validation_chain.invoke({
            "question": question, 
            "context": db_context
        }).strip().lower()

        is_sufficient = response == "true"  # 응답이 'true'이면 충분하다고 판단합니다.
        
        if is_sufficient:
            print("✅ 검색된 문서만으로 답변이 충분합니다.")  # 충분하다고 판단된 경우 메시지 출력
        else:
            print("⚠️ 검색된 문서가 불충분합니다. 웹 검색을 시도합니다.")  # 불충분하다고 판단된 경우 메시지 출력
            
    except Exception as e:
        print(f"❌ LLM 검증 실패: {e}. 불충분으로 간주합니다.")  # LLM 호출 실패 시 오류 메시지 출력
        is_sufficient = False  # 오류 발생 시 불충분으로 처리합니다.
        
    return {**state, "is_retrieval_sufficient": is_sufficient}  # 검증 결과를 상태에 저장하고 반환합니다.

# [노드] DB 컨텍스트와 웹 컨텍스트를 결합하여 최종 컨텍스트를 만드는 노드
def combine_context_node(state: GraphState) -> Dict[str, Any]:
    """DB 컨텍스트와 웹 컨텍스트를 결합하여 최종 컨텍스트를 만드는 노드 함수"""
    print("--- 노드: 컨텍스트 결합 ---")  # 현재 노드명을 출력합니다.
    db_context = state.get("db_context", "")  # 상태에서 'db_context' 값을 가져옵니다.
    web_context = state.get("web_context", "")  # 상태에서 'web_context' 값을 가져옵니다.
    final_context = db_context  # 기본 최종 컨텍스트를 DB 컨텍스트로 설정합니다.
    # 웹 컨텍스트가 존재하고 비활성화 상태가 아닐 경우, DB 컨텍스트와 결합합니다.
    if web_context and web_context != "웹 검색 비활성화":
        print("✅ DB와 웹 컨텍스트를 결합합니다.")  # 결합을 알리는 메시지를 출력합니다.
        final_context = f"[DB 검색 결과]\n{db_context}\n\n[웹 검색 결과]\n{web_context}"  # 두 컨텍스트를 결합합니다.
    else:
        print("ℹ️ DB 컨텍스트만 사용합니다.")  # 웹 컨텍스트가 없는 경우 메시지 출력
    return {**state, "context": final_context}  # 결합된 컨텍스트를 상태에 저장하고 반환합니다.

# [노드] 최종 컨텍스트를 기반으로 LLM이 초안 답변을 생성하는 노드
def generate_draft_node(state: GraphState) -> Dict[str, Any]:
    """최종 컨텍스트를 기반으로 LLM이 초안 답변을 생성하는 노드 함수"""
    print("--- 노드: 초안 생성 ---")  # 현재 노드명을 출력합니다.
    context = state.get("context", "")  # 상태에서 'context' 값을 가져옵니다.
    question = state.get("question", "")  # 상태에서 'question' 값을 가져옵니다.
    
    # 컨텍스트가 없으면 답변 불가를 반환합니다.
    if not context:
        print("❌ 컨텍스트가 없어 답변을 생성할 수 없습니다.")  # 오류 메시지를 출력합니다.
        return {**state, "answer_draft": "주어진 정보로는 답변할 수 없습니다."}  # 답변을 해당 메시지로 설정합니다.

    # RAG 프롬프트를 사용하여 LLM을 호출하고 답변을 생성합니다.
    response = llm.invoke(rag_prompt.format(context=context, question=question))
    ans = response.content  # LLM 응답의 내용을 추출합니다.
    
    print("✅ 답변 초안 생성 완료.")  # 초안 생성 완료 메시지 출력
    return {**state, "answer_draft": ans, "answer_source": "내부 DB/웹 검색"}  # 생성된 초안과 출처를 상태에 저장하고 반환합니다.

# [노드] 초안 답변을 최종 답변으로 확정하는 노드
def refine_answer_node(state: GraphState) -> Dict[str, Any]:
    """초안 답변을 최종 답변으로 확정하는 노드 함수"""
    print("--- 노드: 답변 개선 및 최종 생성 ---")  # 현재 노드명을 출력합니다.
    answer = state.get("answer_draft", "")  # 상태에서 'answer_draft' 값을 가져옵니다.
    print("✅ 답변 개선 완료.")  # 답변 개선 완료 메시지 출력
    return {**state, "answer": answer}  # 초안을 최종 답변으로 상태에 저장하고 반환합니다.

# [노드] LLM을 사용하여 생성된 답변의 품질을 최종 검증하는 노드
def answer_validation_node(state: GraphState) -> Dict[str, Any]:
    """LLM을 사용하여 생성된 답변의 품질을 최종 검증하는 노드 함수"""
    print("--- 노드: 2차 검증 (답변 품질) ---")  # 현재 노드명을 출력합니다.
    question = state.get("question")  # 상태에서 'question' 값을 가져옵니다.
    context = state.get("context", "")  # 상태에서 'context' 값을 가져옵니다.
    answer = state.get("answer", "")  # 상태에서 'answer' 값을 가져옵니다.

    # 답변 검증용 LLM 체인을 정의합니다.
    validation_chain = ChatPromptTemplate.from_template(VALIDATION_PROMPT_TMPL) | llm | StrOutputParser()

    try:
        # LLM을 호출하여 답변이 충분한지 검증합니다.
        response = validation_chain.invoke({"question": question, "context": context, "answer": answer}).strip().lower()
        is_sufficient = response == "true"  # 응답이 'true'이면 충분하다고 판단합니다.
        
        if is_sufficient:
            print("✅ 답변이 질문에 충분히 충실하고 관련성이 높습니다.")  # 충분하다고 판단된 경우 메시지 출력
        else:
            print("⚠️ 답변이 불충분하거나 관련성이 낮습니다.")  # 불충분하다고 판단된 경우 메시지 출력
            
    except Exception as e:
        print(f"❌ LLM 답변 검증 실패: {e}. 불충분으로 간주합니다.")  # LLM 호출 실패 시 오류 메시지 출력
        is_sufficient = False  # 오류 발생 시 불충분으로 처리합니다.
        
    print(f"  - 최종 결과: {'✅ 충분' if is_sufficient else '⚠️ 불충분'}")  # 최종 검증 결과를 출력합니다.
    return {**state, "is_answer_sufficient": is_sufficient}  # 검증 결과를 상태에 저장하고 반환합니다.

# [노드] Tavily API를 사용하여 웹 검색을 수행하는 노드
def web_search_node(state: GraphState) -> Dict[str, Any]:
    """Tavily API를 사용하여 웹 검색을 수행하는 노드 함수"""
    print("--- 노드: 웹 검색 ---")  # 현재 노드명을 출력합니다.
    question = state.get("question")  # 상태에서 'question' 값을 가져옵니다.
    if not question:
        raise ValueError("질문이 누락되었습니다.")  # 질문이 없으면 오류를 발생시킵니다.
    if not TAVILY_API_KEY:
        print("⚠️ TAVILY API 키가 없어 웹 검색을 건너뜁니다.")  # API 키가 없으면 경고 메시지 출력
        return {**state, "web_context": "웹 검색 비활성화"}  # 웹 검색을 비활성화한 상태를 반환합니다.
    
    search_tool = TavilyClient(api_key=TAVILY_API_KEY)  # Tavily 클라이언트 인스턴스를 생성합니다.
    web_context_parts = []  # 웹 검색 결과를 저장할 리스트를 초기화합니다.
    
    try:
        # Tavily 검색 클라이언트를 사용하여 질문에 대한 웹 검색을 수행합니다.
        response = search_tool.search(query=question, max_results=5)
        
        if isinstance(response, dict) and "results" in response:
            results = response["results"]
            for r in results:
                if isinstance(r, dict):
                    title = (r.get("title") or "").strip()
                    content = (r.get("content") or r.get("snippet") or "").strip()
                    url = (r.get("url") or "").strip()
                    web_context_parts.append(f"- 출처: {url or 'N/A'}\n 내용: {content}")  # 검색 결과를 리스트에 추가합니다.
        else:
            print(f"⚠️ 예상치 못한 Tavily 응답 형식: {type(response)}")  # 예상치 못한 응답 형식에 대한 경고 출력
            web_context_parts.append(f"- 출처: N/A\n 내용: 웹 검색 응답 형식 오류")
    except Exception as e:
        print(f"❌ Tavily 검색 오류: {e}")  # Tavily 검색 중 오류가 발생하면 오류 메시지 출력
        web_context_parts.append(f"- 출처: N/A\n 내용: 웹 검색 실패 - {str(e)}")

    web_context = "\n\n".join(web_context_parts)  # 리스트에 저장된 웹 검색 결과를 하나의 문자열로 결합합니다.
    print("✅ 웹 검색 완료.")  # 웹 검색 완료 메시지 출력
    return {**state, "web_context": web_context}  # 웹 검색 결과를 상태에 저장하고 반환합니다.

# [노드] 어떤 답변도 생성할 수 없을 때 대체 답변을 반환하는 노드
def fallback_answer_node(state: GraphState) -> Dict[str, Any]:
    """어떤 답변도 생성할 수 없을 때 대체 답변을 반환하는 노드 함수"""
    fallback_msg = "죄송하지만 이 질문에 대한 답변은 제공할 수 없습니다. 다른 질문을 해주세요."
    print("\n" + fallback_msg)  # 대체 메시지를 출력합니다.
    return {**state, "answer": fallback_msg, "answer_source": "Fallback"}  # 대체 메시지를 최종 답변으로 상태에 저장하고 반환합니다.

# ==================== 그래프 빌드 및 실행 ====================
def build_graph():
    """LangGraph 상태 그래프의 노드와 흐름을 정의하고 컴파일하는 함수"""
    g = StateGraph(GraphState)  # GraphState를 기반으로 새로운 그래프를 생성합니다.
    
    # 그래프에 각 노드를 추가합니다.
    g.add_node("load_milvus", load_milvus_node)  # Milvus 로드 노드 추가
    g.add_node("retrieve", retrieve_node)  # 문서 검색 노드 추가
    g.add_node("retrieval_validation", retrieval_validation_node)  # 검색 품질 검증 노드 추가
    g.add_node("web_search", web_search_node)  # 웹 검색 노드 추가
    g.add_node("combine_context", combine_context_node)  # 컨텍스트 결합 노드 추가
    g.add_node("generate_draft", generate_draft_node)  # 초안 답변 생성 노드 추가
    g.add_node("refine_answer", refine_answer_node)  # 답변 개선 노드 추가
    g.add_node("answer_validation", answer_validation_node)  # 답변 품질 검증 노드 추가
    g.add_node("fallback_answer", fallback_answer_node)  # 대체 답변 노드 추가

    # 그래프의 시작점을 'load_milvus' 노드로 설정합니다.
    g.set_entry_point("load_milvus")
    # 'load_milvus' 노드에서 'retrieve' 노드로 이동하는 엣지를 추가합니다.
    g.add_edge("load_milvus", "retrieve")
    # 'retrieve' 노드에서 'retrieval_validation' 노드로 이동하는 엣지를 추가합니다.
    g.add_edge("retrieve", "retrieval_validation")

    # 1차 검증 결과에 따라 다음 노드를 조건부로 결정합니다.
    g.add_conditional_edges(
        "retrieval_validation",
        lambda state: "sufficient" if state.get("is_retrieval_sufficient") else "insufficient",
        {"sufficient": "combine_context", "insufficient": "web_search"}
    )
    # 웹 검색 후 'combine_context' 노드로 이동하는 엣지를 추가합니다.
    g.add_edge("web_search", "combine_context")
    # 컨텍스트 결합 후 답변 초안 생성 노드로 이동합니다.
    g.add_edge("combine_context", "generate_draft")
    # 답변 초안 생성 후 개선 노드로 이동합니다.
    g.add_edge("generate_draft", "refine_answer")
    # 답변 개선 후 최종 검증 노드로 이동합니다.
    g.add_edge("refine_answer", "answer_validation")

    # 2차 검증 결과에 따라 최종 흐름을 결정하는 함수를 정의합니다.
    def decide_after_answer_validation(state: GraphState) -> str:
        if state.get("is_answer_sufficient"):
            return "end"  # 답변이 충분하면 그래프를 종료합니다.
        else:
            print(f"❌ 답변 불충분. 대체 답변으로 이동.")
            return "fallback"  # 답변이 불충분하면 'fallback' 상태를 반환합니다.

    # 2차 검증 결과에 따라 조건부 엣지를 추가합니다.
    g.add_conditional_edges(
        "answer_validation",
        decide_after_answer_validation,
        {"end": END, "fallback": "fallback_answer"}
    )
    # 대체 답변 노드에서 그래프를 종료하는 엣지를 추가합니다.
    g.add_edge("fallback_answer", END)
    
    # 정의된 그래프를 컴파일하여 실행 가능한 객체를 반환합니다.
    return g.compile()

async def run(state: dict) -> dict:
    """오케스트레이터에서 호출될 메인 비동기 실행 함수"""
    try:
        query = state.get("query", "")  # 상태 딕셔너리에서 'query' 값을 가져옵니다.
        if not query:
            return {"agent_answer": "질문이 제공되지 않았습니다. 작물추천 관련 질문을 해주세요."}  # 질문이 없으면 오류 메시지 반환
        print(f"[작물추천_agent] 질문 처리 시작: {query}")  # 질문 처리 시작을 알립니다.
        app = build_graph()  # 그래프를 빌드하고 컴파일합니다.
        final_state = app.invoke({"question": query})  # LangGraph를 호출하고 최종 상태를 가져옵니다.
        
        if isinstance(final_state, dict):
            answer = final_state.get("answer", "답변 생성에 실패했습니다.")  # 최종 상태에서 답변을 가져옵니다.
        elif isinstance(final_state, str):
            answer = final_state
        else:
            answer = "답변 형식이 올바르지 않습니다."
        
        print(f"[작물추천_agent] 답변 생성 완료: {len(answer)}자")  # 생성된 답변의 길이를 출력합니다.
        return {"agent_answer": answer}  # 최종 답변을 딕셔너리 형태로 반환합니다.
    except Exception as e:
        error_msg = f"작물추천 에이전트 실행 중 오류가 발생했습니다: {e}"
        print(f"[작물추천_agent] 오류: {e}")
        return {"agent_answer": error_msg}

def remove_markdown_and_special_chars(text: str) -> str:
    """Markdown 문법과 특수 문자를 제거하여 깨끗한 텍스트를 반환하는 함수"""
    # Markdown 헤더, 리스트, 강조(볼드체) 등을 제거하는 정규식
    text = re.sub(r'#{1,6}\s', '', text)  # '# ' 형식의 Markdown 헤더를 제거합니다.
    text = re.sub(r'[\*\-]', '', text)  # 별표(*), 하이픈(-)과 같은 특수 문자를 제거합니다.
    text = re.sub(r'\[.*?\]\(.*?\)', '', text)  # Markdown 링크([텍스트](URL))를 제거합니다.
    text = re.sub(r'\s+', ' ', text)  # 여러 개의 공백을 하나로 축소하여 텍스트를 정리합니다.
    
    return text.strip()  # 문자열의 양 끝에 있는 공백을 제거하고 반환합니다.
